function Format-Size([long]$Bytes) {
    if ($Bytes -lt 1024) { return "$Bytes B" }
    if ($Bytes -lt 1MB) { return ('{0} KB' -f [long]($Bytes / 1KB)) }
    if ($Bytes -lt 1GB) { return ('{0:F2} MB' -f ($Bytes / 1MB)) }
    return ('{0:F2} GB' -f ($Bytes / 1GB))
}

function Find-JsonKey {
    param($Object, [string]$Key)
    if ($Object -is [System.Management.Automation.PSCustomObject]) {
        if ($Object.PSObject.Properties.Name -contains $Key) { return $Object.$Key }
        foreach ($p in $Object.PSObject.Properties) {
            $r = Find-JsonKey $p.Value $Key
            if ($null -ne $r) { return $r }
        }
    }
    elseif ($Object -is [System.Collections.IEnumerable] -and $Object -isnot [string]) {
        foreach ($item in $Object) {
            $r = Find-JsonKey $item $Key
            if ($null -ne $r) { return $r }
        }
    }
    return $null
}

function ConvertTo-PathCode([string]$Text) {
    $sb = New-Object System.Text.StringBuilder
    foreach ($b in [System.Text.Encoding]::UTF8.GetBytes($Text)) {
        if (($b -ge 0x30 -and $b -le 0x39) -or ($b -ge 0x41 -and $b -le 0x5A) -or ($b -ge 0x61 -and $b -le 0x7A)) {
            [void]$sb.Append([char]$b)
        }
        else {
            [void]$sb.Append(('%{0:X2}' -f $b))
        }
    }
    return $sb.ToString()
}

function Invoke-ShareApi {
    param(
        [string]$Uri,
        [string]$Method = 'Get',
        $Session,
        [string]$Referer,
        $Body = $null
    )
    $splat = @{
        UseBasicParsing = $true
        Uri             = $Uri
        Method          = $Method
        Headers         = @{ Referer = $Referer; Accept = 'application/json' }
        UserAgent       = $UA
        WebSession      = $Session
    }
    if ($Method -ne 'Get') { $splat.ContentType = 'application/json' }
    if ($null -ne $Body) { $splat.Body = $Body }
    try {
        return Invoke-RestMethod @splat
    }
    catch {
        $detail = $null
        if ($_.ErrorDetails) { $detail = $_.ErrorDetails.Message }
        if ($detail) { throw "HTTP 请求失败 ($Method $Uri)`nAPI 响应: $detail" }
        throw
    }
}

function Get-RemoteSize {
    param([string]$Url, [string]$Referer)
    $resp = $null
    try {
        $req = [System.Net.HttpWebRequest]::Create($Url)
        $req.UserAgent = $UA
        $req.Referer = $Referer
        $req.Method = 'GET'
        $req.Timeout = 30000
        $req.AddRange(0, 0)
        $resp = $req.GetResponse()
        if ([int]$resp.StatusCode -eq 206) {
            $cr = $resp.Headers['Content-Range']
            if ($cr -and $cr -match '/(\d+)\s*$') { return [long]$Matches[1] }
        }
        if ($resp.ContentLength -gt 0) { return [long]$resp.ContentLength }
    }
    catch { }
    finally {
        if ($resp) { $resp.Close() }
    }
    return -1
}

function Save-ToFile {
    param([string]$Url, [string]$Referer, [string]$Destination, [long]$Total)

    $req = [System.Net.HttpWebRequest]::Create($Url)
    $req.UserAgent = $UA
    $req.Referer = $Referer
    $req.Method = 'GET'
    $req.AllowAutoRedirect = $true
    $req.Timeout = 30000
    $req.ReadWriteTimeout = 60000

    $resp = $null
    $stream = $null
    $fs = $null
    [long]$done = 0
    try {
        $resp = $req.GetResponse()
        if ($Total -le 0 -and $resp.ContentLength -gt 0) { $Total = [long]$resp.ContentLength }
        $stream = $resp.GetResponseStream()
        $fs = [System.IO.File]::Create($Destination)
        $buf = New-Object byte[] (256 * 1024)
        $start = Get-Date
        $lastPrint = Get-Date
        while ($true) {
            $n = $stream.Read($buf, 0, $buf.Length)
            if ($n -le 0) { break }
            $fs.Write($buf, 0, $n)
            $done += $n
            $now = Get-Date
            if (($now - $lastPrint).TotalSeconds -ge 0.5) {
                $lastPrint = $now
                $elapsed = [math]::Max((($now - $start).TotalSeconds), 0.001)
                $speed = Format-Size ([long]($done / $elapsed))
                if ($Total -gt 0) {
                    $line = ('{0,3}%  {1} / {2}  {3}/s' -f [int](($done * 100) / $Total), (Format-Size $done), (Format-Size $Total), $speed)
                }
                else {
                    $line = ('{0}  {1}/s' -f (Format-Size $done), $speed)
                }
                Write-Host ("`r" + $line.PadRight(60)) -NoNewline
            }
        }
        $fs.Flush()
    }
    finally {
        if ($fs) { $fs.Dispose() }
        if ($stream) { $stream.Dispose() }
        if ($resp) { $resp.Close() }
    }
    Write-Host ("`r" + (' ' * 60)) -NoNewline
    Write-Host ''
    if ($Total -gt 0 -and $done -ne $Total) {
        throw "下载不完整: $done / $Total bytes"
    }
}
