function Format-Size([double]$b) {
    if ($b -ge $script:GB) { return ('{0:F2} GB' -f ($b / $script:GB)) }
    if ($b -ge $script:MB) { return ('{0:F2} MB' -f ($b / $script:MB)) }
    if ($b -ge $script:KB) { return ('{0:F1} KB' -f ($b / $script:KB)) }
    return ('{0} B' -f [long]$b)
}

function Format-Speed([double]$bps) {
    if ($bps -le 0) { return '0 B/s' }
    return (Format-Size $bps) + '/s'
}

function Format-Eta([double]$sec) {
    if ($sec -le 0 -or $sec -gt 359999) { return '' }
    if ($sec -lt 60) { return ('{0:F0} 秒' -f $sec) }
    if ($sec -lt 3600) { return ('{0:F0} 分 {1:F0} 秒' -f [Math]::Floor($sec / 60), ($sec % 60)) }
    return ('{0:F0} 时 {1:F0} 分' -f [Math]::Floor($sec / 3600), [Math]::Floor(($sec % 3600) / 60))
}

function ConvertTo-Bytes([string]$s) {
    if ([string]::IsNullOrWhiteSpace($s)) { return [long]0 }
    $m = [regex]::Match($s, '([0-9]+(?:\.[0-9]+)?)\s*([KMG]?)(?:B)?', 'IgnoreCase')
    if (-not $m.Success) { return [long]0 }
    $v = [double]$m.Groups[1].Value
    switch ($m.Groups[2].Value.ToUpper()) {
        'G' { $v = $v * 1073741824 }
        'M' { $v = $v * 1048576 }
        'K' { $v = $v * 1024 }
    }
    return [long]$v
}

function Clean-Name([string]$s) {

function Fix-SplitName([string]$name) {
    if ($name -match '^(.+)\.([0-9]{3})\.7z$') {
        return ($matches[1] + '.7z.' + $matches[2])
    }
    return $name
}

function ConvertTo-SafeName([string]$name) {
    $n = $name -replace '[\\/:*?"<>|]', '_'
    if ([string]::IsNullOrWhiteSpace($n)) { $n = 'download.bin' }
    return $n.Trim()
}

function Resolve-Lanzou {

    $req.ContentType = 'application/json'

    $req.UserAgent = $script:UserAgent
    $req.Timeout = 20000
    $req.ContentLength = $bytes.Length

    $text = ''
    try {
        $st = $req.GetRequestStream()
        $st.Write($bytes, 0, $bytes.Length)
        $st.Dispose()
        $resp = $req.GetResponse()
        $sr = New-Object System.IO.StreamReader($resp.GetResponseStream(), [System.Text.Encoding]::UTF8)
        $text = $sr.ReadToEnd()
        $sr.Dispose()
        $resp.Dispose()
    } catch {
        $msg = $_.Exception.Message
        try {
            if ($_.Exception.Response) {
                $sr2 = New-Object System.IO.StreamReader($_.Exception.Response.GetResponseStream(), [System.Text.Encoding]::UTF8)
                $body = $sr2.ReadToEnd(); $sr2.Dispose()
                if ($body) { $msg = $body }
            }
        } catch { }
        throw "接口请求失败: $msg"
    }

    try { $j = $text | ConvertFrom-Json }
    catch { throw "解析 API 响应失败: $($text.Substring(0, [Math]::Min(200, $text.Length)))" }

    if ($j.code -ne 0) {
        $m = [string]$j.msg

function Get-RemoteInfo {
    param([string]$Link, [string]$Referer)

    $r = [pscustomobject]@{ Size = [long]0; SupportRange = $false }
    $req = [System.Net.HttpWebRequest]::Create($Link)
    $req.Method = 'GET'
    $req.UserAgent = $script:UserAgent
    if ($Referer) { $req.Referer = $Referer }
    $req.Accept = '*/*'
    $req.Timeout = 20000
    $req.ReadWriteTimeout = 20000
    [void]$req.AddRange(0, 0)

    try {
        $resp = $req.GetResponse()
        $code = [int]$resp.StatusCode
        if ($code -eq 206) {
            $cr = [string]$resp.Headers['Content-Range']
            if ($cr -match '/\s*([0-9]+)') {
                $r.Size = [long]$matches[1]
                $r.SupportRange = $true
            }
        } elseif ($code -eq 200) {

            $cl = [string]$resp.Headers['Content-Length']
            if ($cl) { $r.Size = [long]$cl }
            $r.SupportRange = $false
        }
        $resp.Dispose()
    } catch {
        try { if ($_.Exception.Response) { $_.Exception.Response.Dispose() } } catch { }
    }
    return $r
}

function Get-BlockLength([int]$Index, [long]$Total, [long]$Block) {
    $s = [long]$Index * $Block
    $rest = $Total - $s
    if ($rest -gt $Block) { return $Block }
    return $rest
}

function Save-Parallel {
    param($Info, [string]$Dest, [long]$Total, [int]$Threads, [int]$Retry, [bool]$Resume)

    $block = [long]$script:BlockSize
    $count = [int][Math]::Ceiling([double]$Total / [double]$block)
    $tmpDir = "$Dest.part.blocks"
    $statePath = "$Dest.part.json"

    $validState = $false
    if ($Resume -and (Test-Path -LiteralPath $statePath)) {
        try {
            $st = Get-Content -LiteralPath $statePath -Raw | ConvertFrom-Json
            if ([long]$st.total -eq $Total -and [long]$st.block -eq $block) { $validState = $true }
        } catch { }
    }
    if (-not $validState) {
        if (Test-Path -LiteralPath $tmpDir) { Remove-Item -LiteralPath $tmpDir -Recurse -Force }
    }
    if (-not (Test-Path -LiteralPath $tmpDir)) { New-Item -ItemType Directory -Path $tmpDir -Force | Out-Null }
    @{ total = $Total; block = $block; url = $Info.DirectUrl } | ConvertTo-Json -Compress |
        Set-Content -LiteralPath $statePath -Encoding ASCII

    $doneBytes = [long]0
    $todo = New-Object System.Collections.Generic.List[int]
    for ($i = 0; $i -lt $count; $i++) {
        $bp = Join-Path $tmpDir "$i.blk"
        $expect = Get-BlockLength $i $Total $block
        $ok = $false
        if (Test-Path -LiteralPath $bp) {
            if ((Get-Item -LiteralPath $bp).Length -eq $expect) { $ok = $true }
            else { Remove-Item -LiteralPath $bp -Force -ErrorAction SilentlyContinue }
        }
        if ($ok) { $doneBytes += $expect } else { $todo.Add($i) | Out-Null }
    }

    $startTime = Get-Date
    $lastTime = $startTime
    $lastBytes = $doneBytes
    $inst = 0.0
    $errors = New-Object System.Collections.Generic.List[string]

    Write-Host ('  共 {0} 块 / {1} 线程, 已续传 {2}' -f $count, $Threads, (Format-Size $doneBytes))

    $pool = $null
    $running = New-Object System.Collections.Generic.List[object]
    $ptr = 0
    try {
        if ($todo.Count -gt 0) {
            $pool = [runspacefactory]::CreateRunspacePool(1, $Threads)
            $pool.Open()

            while ($ptr -lt $todo.Count -or $running.Count -gt 0) {
                while ($ptr -lt $todo.Count -and $running.Count -lt $Threads) {
                    $i = $todo[$ptr]; $ptr++
                    $bp = Join-Path $tmpDir "$i.blk"
                    $s = [long]$i * $block
                    $e = $s + (Get-BlockLength $i $Total $block) - 1

                    $ps = [powershell]::Create()
                    $ps.RunspacePool = $pool
                    [void]$ps.AddScript($script:BlockWorker)
                    [void]$ps.AddParameter('Link', $Info.DirectUrl)
                    [void]$ps.AddParameter('Referer', $Info.Referer)
                    [void]$ps.AddParameter('Ua', $script:UserAgent)
                    [void]$ps.AddParameter('Path', $bp)
                    [void]$ps.AddParameter('Start', $s)
                    [void]$ps.AddParameter('End', $e)
                    [void]$ps.AddParameter('Retry', $Retry)

                    $h = $ps.BeginInvoke()
                    $running.Add([pscustomobject]@{ Ps = $ps; Handle = $h; Index = $i }) | Out-Null
                }

                $still = New-Object System.Collections.Generic.List[object]
                foreach ($j in $running) {
                    if ($j.Handle.IsCompleted) {
                        try {
                            $out = $j.Ps.EndInvoke($j.Handle)
                            $msg = ''
                            if ($out -ne $null -and $out.Count -gt 0) { $msg = [string]$out[0] }
                            if ($msg) { $errors.Add("分块 $($j.Index): $msg") }
                            else { $doneBytes += (Get-BlockLength $j.Index $Total $block) }
                        } catch {
                            $errors.Add("分块 $($j.Index): $($_.Exception.Message)")
                        }
                        try { $j.Ps.Dispose() } catch { }
                    } else {
                        $still.Add($j) | Out-Null
                    }
                }
                $running = $still

                $now = Get-Date
                $cur = $doneBytes
                foreach ($j in $running) {
                    $bp = Join-Path $tmpDir "$($j.Index).blk"
                    $len = [long]0
                    try { $len = (Get-Item -LiteralPath $bp -ErrorAction Stop).Length } catch { $len = 0 }
                    $cap = Get-BlockLength $j.Index $Total $block
                    if ($len -gt $cap) { $len = $cap }
                    $cur += $len
                }
                if ($cur -gt $Total) { $cur = $Total }

                $elapsed = ($now - $startTime).TotalSeconds
                $avg = 0.0
                if ($elapsed -gt 0.15) { $avg = $cur / $elapsed }
                $dt = ($now - $lastTime).TotalSeconds
                if ($dt -ge 0.5) {
                    if ($cur -ge $lastBytes) { $inst = ($cur - $lastBytes) / $dt }
                    $lastTime = $now
                    $lastBytes = $cur
                }

                $pct = 0
                if ($Total -gt 0) { $pct = [int][Math]::Floor($cur * 100 / $Total) }
                if ($pct -gt 100) { $pct = 100 }
                $eta = ''
                if ($Total -gt $cur -and $avg -gt 1024) {
                    $eta = Format-Eta (($Total - $cur) / $avg)
                }
                $status = '{0} / {1} · {2} · 平均 {3}' -f (Format-Size $cur), (Format-Size $Total), (Format-Speed $inst), (Format-Speed $avg)
                if ($eta) { $status += " · 剩余 $eta" }
                Write-Progress -Activity "下载 $($Info.Name)" -Status $status -PercentComplete $pct -Id 1

                if ($running.Count -gt 0) { Start-Sleep -Milliseconds 150 }
            }
        }
    } finally {
        foreach ($j in $running) { try { $j.Ps.Dispose() } catch { } }
        if ($pool -ne $null) { try { $pool.Close(); $pool.Dispose() } catch { } }
        Write-Progress -Activity "下载 $($Info.Name)" -Completed -Id 1
    }

    if ($errors.Count -gt 0) {
        throw ("有 {0} 个分块失败 (已下载部分保留, 重跑可续传):`n{1}" -f $errors.Count, ($errors -join "`n"))
    }

    $part = "$Dest.part"
    if (Test-Path -LiteralPath $part) { Remove-Item -LiteralPath $part -Force }
    $outFs = [System.IO.File]::Create($part)
    try {
        for ($i = 0; $i -lt $count; $i++) {
            $bfs = [System.IO.File]::OpenRead((Join-Path $tmpDir "$i.blk"))
            $bfs.CopyTo($outFs)
            $bfs.Dispose()
        }
    } finally {
        $outFs.Dispose()
    }

    $real = (Get-Item -LiteralPath $part).Length
    if ($real -ne $Total) {
        throw "合并后大小不符: $real / $Total (保留 .part 便于排查)"
    }

    Move-Item -LiteralPath $part -Destination $Dest -Force
    Remove-Item -LiteralPath $tmpDir -Recurse -Force
    Remove-Item -LiteralPath $statePath -Force -ErrorAction SilentlyContinue

    Write-Host ('  完成: {0}' -f (Format-Size $Total)) -ForegroundColor Green
}

    $req.Accept = '*/*'
    $req.Timeout = 30000
    $req.ReadWriteTimeout = 60000

    $resp = $req.GetResponse()
    $total = [long]$resp.ContentLength
    if ($total -le 0) { $total = $SizeHint }

    $buf = New-Object byte[] 131072
    $cur = [long]0

    $lastTime = $startTime
    $inst = 0.0

            $fs.Write($buf, 0, $n)

                $inst = $n / $dt
                $lastTime = $now

function Format-Size([long]$Bytes) {
    if ($Bytes -lt 1024) { return "$Bytes B" }
    if ($Bytes -lt 1MB) { return ('{0} KB' -f [long]($Bytes / 1KB)) }
    if ($Bytes -lt 1GB) { return ('{0:F2} MB' -f ($Bytes / 1MB)) }
    return ('{0:F2} GB' -f ($Bytes / 1GB))
}

function Find-JsonKey {
    param($Object, [string]$Key)
    if ($Object -is [System.Management.Automation.PSCustomObject]) {
        if ($Object.PSObject.Properties.Name -contains $Key) { return $Object.$Key }
        foreach ($p in $Object.PSObject.Properties) {
            $r = Find-JsonKey $p.Value $Key
            if ($null -ne $r) { return $r }
        }
    }
    elseif ($Object -is [System.Collections.IEnumerable] -and $Object -isnot [string]) {
        foreach ($item in $Object) {
            $r = Find-JsonKey $item $Key
            if ($null -ne $r) { return $r }
        }
    }
    return $null
}

function ConvertTo-PathCode([string]$Text) {
    $sb = New-Object System.Text.StringBuilder
    foreach ($b in [System.Text.Encoding]::UTF8.GetBytes($Text)) {
        if (($b -ge 0x30 -and $b -le 0x39) -or ($b -ge 0x41 -and $b -le 0x5A) -or ($b -ge 0x61 -and $b -le 0x7A)) {
            [void]$sb.Append([char]$b)
        }
        else {
            [void]$sb.Append(('%{0:X2}' -f $b))
        }
    }
    return $sb.ToString()
}

        Method          = $Method
        Headers         = @{ Referer = $Referer; Accept = 'application/json' }
        UserAgent       = $UA
        WebSession      = $Session
    }
    if ($Method -ne 'Get') { $splat.ContentType = 'application/json' }
    if ($null -ne $Body) { $splat.Body = $Body }
    try {
        return Invoke-RestMethod @splat
    }
    catch {
        $detail = $null
        if ($_.ErrorDetails) { $detail = $_.ErrorDetails.Message }
        if ($detail) { throw "HTTP 请求失败 ($Method $Uri)`nAPI 响应: $detail" }
        throw
    }
}

function Get-ApiErrMsg([string]$Body) {
    try {
        $j = $Body | ConvertFrom-Json
        if ($j.msg) { return [string]$j.msg }
    }
    catch { }
    return $Body
}
